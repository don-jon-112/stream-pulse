/* --------------------------------------------------------------------------
   [BAGIAN 2: CUSTOM JAVASCRIPT]
   Salin (Copy) seluruh kode JavaScript di bawah ini, lalu tempel (Paste) 
   ke kolom "Custom JS" di StreamPulse Overlay Studio (http://localhost:3000/studio).
   -------------------------------------------------------------------------- */

(function initJooNathanChatOverlay() {
  const container = document.getElementById('overlayChatContainer');
  if (!container) return;

  // Helper membuat avatar inisial SVG bersih (100% aman, zero-network, anti-broken image / 404)
  function getFallbackAvatar(name, bgColor = '#FE2C55') {
    const char = (name ? name.trim().charAt(0) : '★').toUpperCase() || '★';
    return `data:image/svg+xml,%3Csvg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 100 100'%3E%3Ccircle cx='50' cy='50' r='50' fill='${encodeURIComponent(bgColor)}'/%3E%3Ctext x='50' y='64' font-family='Montserrat, sans-serif' font-size='44' font-weight='800' fill='%23ffffff' text-anchor='middle'%3E${encodeURIComponent(char)}%3C/text%3E%3C/svg%3E`;
  }

  // MutationObserver untuk mendeteksi pesan chat baru dan menerapkan efek typing indicator
  const observer = new MutationObserver((mutations) => {
    mutations.forEach((mutation) => {
      mutation.addedNodes.forEach((node) => {
        if (!node || node.nodeType !== 1 || !node.classList.contains('overlay-item')) return;
        if (node.dataset.jooProcessed) return;
        node.dataset.jooProcessed = 'true';

        // ========================================================
        // 1. TRANSFORMASI ALERT KE FORMAT EXACT .alert-box
        // ========================================================
        if (node.classList.contains('overlay-event')) {
          const authorEl = node.querySelector('.overlay-author-name');
          const titleEl = node.querySelector('.overlay-event-title');
          const msgEl = node.querySelector('.overlay-text');
          const avatarEl = node.querySelector('.overlay-avatar');

          const authorName = (authorEl ? authorEl.textContent.trim() : 'Viewer').replace(/:\s*$/, '');
          const fullTitle = titleEl ? titleEl.textContent.trim() : 'Alert';
          const userMsg = msgEl ? msgEl.textContent.replace(/^"|"$/g, '').trim() : '';

          const isTwitch = node.classList.contains('platform-twitch') || node.dataset.platform === 'twitch';
          const isYouTube = node.classList.contains('platform-youtube') || node.dataset.platform === 'youtube';
          const isTikTok = node.classList.contains('platform-tiktok') || node.dataset.platform === 'tiktok';

          let borderColor = '#9146FF';
          if (isTikTok) borderColor = '#FE2C55';
          if (isYouTube) borderColor = '#FF0000';
          let alertText = fullTitle;

          const titleLower = fullTitle.toLowerCase();
          if (titleLower.includes('follow')) {
            borderColor = '#FFB700';
            alertText = 'just followed';
          } else if (titleLower.includes('raid')) {
            borderColor = '#FF0000';
            alertText = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : fullTitle;
          } else if (titleLower.includes('cheer') || titleLower.includes('bits')) {
            borderColor = '#9C3EE8';
            const bitsAmount = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : 'Bits';
            alertText = `cheered with ${bitsAmount} 🎉`;
          } else if (titleLower.includes('super chat')) {
            borderColor = '#E6A01A';
            const amount = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : '';
            alertText = `sent a Super Chat ${amount} 🎉`;
          } else if (titleLower.includes('member') || titleLower.includes('sponsor')) {
            borderColor = '#FF0000';
            alertText = fullTitle.includes('•') ? `became a member! ${fullTitle.split('•')[1].trim()}` : 'became a member! ⭐';
          } else if (titleLower.includes('gift')) {
            borderColor = isTikTok ? '#FE2C55' : '#FF1493';
            const giftDetail = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : '';
            alertText = `gifted ${giftDetail} 🎁`;
          } else if (titleLower.includes('like')) {
            borderColor = '#FF4D6D';
            alertText = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : 'sent likes! ❤️';
          } else if (titleLower.includes('sub')) {
            borderColor = '#9146FF';
            alertText = fullTitle.includes('•') ? `subscribed at ${fullTitle.split('•')[1].trim()}` : 'subscribed at Tier 1';
          } else if (titleLower.includes('redeem')) {
            borderColor = '#00BFA5';
            const rDetail = fullTitle.includes('•') ? fullTitle.split('•')[1].trim() : 'Channel Points';
            alertText = `redeemed ${rDetail} 💠`;
          }

          const fallbackAvatar = getFallbackAvatar(authorName, borderColor);
          let avatarSrc = '';
          if (avatarEl) {
            if (avatarEl.tagName === 'IMG' && avatarEl.src && !avatarEl.src.includes('default-profile_image') && !avatarEl.src.endsWith('/undefined')) {
              avatarSrc = avatarEl.src;
            }
          }
          if (!avatarSrc) {
            avatarSrc = fallbackAvatar;
          }

          let platformIcon = '';
          if (isYouTube) {
            platformIcon = '<img src="https://i.imgur.com/cbl1elx.png" style="height: 1.2em; width: auto; vertical-align: middle; margin: 0 4px;" alt="YouTube">';
          } else if (isTwitch) {
            platformIcon = '<img src="https://i.imgur.com/vLsqr6V.png" style="height: 1.2em; width: auto; vertical-align: middle; margin: 0 4px;" alt="Twitch">';
          } else if (isTikTok) {
            platformIcon = '<span style="display:inline-block; margin:0 4px; font-weight:800; font-size:0.9em; color:#fe2c55;">♪</span>';
          }

          node.innerHTML = `
            <div class="alert-box" style="border-color: ${borderColor} !important;">
              <img src="${avatarSrc}" class="alert-avatar" style="border-color: ${borderColor} !important;" onerror="this.onerror=null;this.src='${fallbackAvatar}';">
              <div class="alert-text">
                <span style="color: ${borderColor} !important; font-weight: 700;">${authorName}</span>${platformIcon}<span>${alertText}</span>
                ${userMsg ? `<span class="alert-user-msg">"${userMsg}"</span>` : ''}
              </div>
            </div>
          `;

          // Jika avatar dari Twitch dan masih fallback, coba resolve dari Twitch API
          if (isTwitch && authorName && authorName !== 'Viewer') {
            fetch(`/api/twitch/avatar/${encodeURIComponent(authorName)}`)
              .then(r => r.json())
              .then(res => {
                if (res && res.avatar) {
                  const img = node.querySelector('.alert-avatar');
                  if (img) img.src = res.avatar;
                }
              })
              .catch(() => {});
          }
          return;
        }

        const textContainer = node.querySelector('.overlay-text');
        const contentSpan = node.querySelector('.overlay-message-content') || textContainer;
        if (!textContainer || !contentSpan) return;

        // Bersihkan tanda titik dua (:) setelah nama user jika ada
        const authorEl = node.querySelector('.overlay-author-name');
        if (authorEl && authorEl.textContent.endsWith(':')) {
          authorEl.textContent = authorEl.textContent.replace(/:\s*$/, '');
        }

        // Warna border avatar: gunakan warna profil/username jika pesan dari Twitch, tetap default putih jika dari platform lain (YouTube, TikTok, dll)
        const isTwitch = node.classList.contains('platform-twitch') || node.dataset.platform === 'twitch';
        const avatarEl = node.querySelector('.overlay-avatar');
        if (avatarEl) {
          if (isTwitch && authorEl && authorEl.style.color) {
            avatarEl.style.setProperty('border-color', authorEl.style.color, 'important');
          } else {
            avatarEl.style.setProperty('border-color', '#ffffff', 'important');
          }
        }

        // Simpan teks asli dan sembunyikan sementara
        const originalContent = contentSpan.innerHTML;
        contentSpan.style.display = 'none';

        // Buat Typing Indicator (3 Bouncing Dots)
        const typingEl = document.createElement('div');
        typingEl.className = 'typing-indicator';
        typingEl.innerHTML = '<div class="dot"></div><div class="dot"></div><div class="dot"></div>';
        textContainer.insertBefore(typingEl, contentSpan);

        // Setelah 1.2 detik, hilangkan typing indicator dan tampilkan pesan teks
        setTimeout(() => {
          if (typingEl.parentNode) {
            typingEl.remove();
          }
          contentSpan.style.display = 'inline-block';
          contentSpan.style.animation = 'jooFadeIn 0.3s ease forwards';
        }, 1200);
      });
    });
  });

  observer.observe(container, { childList: true });

  // Tambahkan keyframe animasi fade-in dinamis untuk teks pesan
  if (!document.getElementById('joo-dynamic-anim-styles')) {
    const style = document.createElement('style');
    style.id = 'joo-dynamic-anim-styles';
    style.textContent = `
      @keyframes jooFadeIn {
        from { opacity: 0; transform: translateY(4px); }
        to { opacity: 1; transform: translateY(0); }
      }
    `;
    document.head.appendChild(style);
  }

  console.log('[StreamPulse] Joo_Nathan Custom Chat Overlay initialized successfully!');
})();
