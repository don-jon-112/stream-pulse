/* --------------------------------------------------------------------------
   [BAGIAN 2: CUSTOM JAVASCRIPT]
   Salin seluruh kode JS di bawah ini lalu tempel ke kolom:
   "Custom JavaScript (Opsional)" di StreamPulse Overlay Studio
   -------------------------------------------------------------------------- */

(function initCartoonThoughtBubble(config, socket) {
  // Palet 11 warna cerah kartun
  const CARTOON_PALETTE = [
    '#00bfa5', // Mint
    '#f43f5e', // Hot Pink
    '#6366f1', // Indigo Purple
    '#0ea5e9', // Sky Blue
    '#f59e0b', // Amber / Orange
    '#8b5cf6', // Violet
    '#10b981', // Emerald
    '#ec4899', // Magenta Pink
    '#ff5722', // Coral Orange
    '#14b8a6', // Deep Teal
    '#84cc16'  // Lime Green
  ];

  let randomPaletteIdx = 0;
  const userColorCache = new Map();

  function toRgba(color, alpha = 0.16) {
    if (!color) return `rgba(0, 191, 165, ${alpha})`;
    if (color.startsWith('rgb(')) {
      return color.replace('rgb(', 'rgba(').replace(')', `, ${alpha})`);
    }
    if (color.startsWith('rgba(')) return color;
    let c = String(color).replace('#', '').trim();
    if (c.length === 3) c = c.split('').map(x => x + x).join('');
    if (c.length === 6) {
      const num = parseInt(c, 16);
      return `rgba(${(num >> 16) & 255}, ${(num >> 8) & 255}, ${num & 255}, ${alpha})`;
    }
    return `rgba(0, 191, 165, ${alpha})`;
  }

  // Pilih warna:
  // 1. Jika Twitch punya warna profil asli, gunakan warna Twitch tersebut!
  // 2. Jika tidak ada (atau YouTube/TikTok), gunakan warna acak dari palet
  function getBubbleThemeColor(itemEl, authorName, authorEl) {
    const isTwitch = itemEl.classList.contains('platform-twitch') ||
                     itemEl.dataset.platform === 'twitch';

    if (isTwitch) {
      if (itemEl.dataset.authorColor && itemEl.dataset.hasCustomColor === 'true') {
        return itemEl.dataset.authorColor;
      }
      if (authorEl) {
        const savedColor = authorEl.dataset.origColor;
        if (savedColor && savedColor !== '#ffffff' && savedColor !== 'rgb(255, 255, 255)' &&
            savedColor !== '#67e8f9' && savedColor !== 'rgb(103, 232, 249)') {
          return savedColor;
        }
        const styleAttr = authorEl.getAttribute('style') || '';
        const match = styleAttr.match(/color\s*:\s*([^;!]+)/i);
        if (match && match[1]) {
          const raw = match[1].trim();
          const lower = raw.toLowerCase();
          if (lower !== '#67e8f9' && lower !== 'rgb(103, 232, 249)' &&
              lower !== '#ffffff' && lower !== 'rgb(255, 255, 255)') {
            return raw;
          }
        }
      }
    }

    const cleanName = String(authorName || '').toLowerCase().trim();
    if (cleanName && userColorCache.has(cleanName)) {
      return userColorCache.get(cleanName);
    }

    const chosenColor = CARTOON_PALETTE[randomPaletteIdx % CARTOON_PALETTE.length];
    randomPaletteIdx++;
    if (cleanName) {
      userColorCache.set(cleanName, chosenColor);
    }
    return chosenColor;
  }

  function applyThoughtBubble(itemEl) {
    if (!itemEl || itemEl._thoughtReady) return;
    itemEl._thoughtReady = true;

    const authorEl = itemEl.querySelector('.overlay-author-name');
    const authorName = authorEl ? authorEl.textContent.trim() : '';
    const headerEl = itemEl.querySelector('.overlay-header');

    // 1. Tentukan platform spesifik (Twitch vs YouTube vs TikTok)
    const isTwitch = itemEl.classList.contains('platform-twitch') || itemEl.dataset.platform === 'twitch';
    const isYouTube = itemEl.classList.contains('platform-youtube') || itemEl.dataset.platform === 'youtube';

    // 2. Warna tema (Twitch custom color jika ada, atau random color)
    const themeColor = getBubbleThemeColor(itemEl, authorName, authorEl);
    const shadowRgba = toRgba(themeColor, 0.22);
    const dotRgba = toRgba(themeColor, 0.16);

    itemEl.style.setProperty('--bubble-theme-color', themeColor);
    itemEl.style.setProperty('--bubble-border', themeColor);
    itemEl.style.setProperty('--pill-bg', themeColor);
    itemEl.style.setProperty('--bubble-shadow', shadowRgba);
    itemEl.style.setProperty('--bubble-dot-color', dotRgba);

    // 3. PAKSA WARNA NAMA PENGIRIM PUTIH MURNI (#ffffff)
    if (authorEl) {
      if (!authorEl.dataset.origColor) {
        authorEl.dataset.origColor = authorEl.style.color || '';
      }
      authorEl.style.setProperty('color', '#ffffff', 'important');
      authorEl.style.setProperty('-webkit-text-fill-color', '#ffffff', 'important');
    }

    // 4. HAPUS SEMUA BADGE BAWAAN PLATFORM, BADGE ALERT KUNING & TAG REDEEM
    if (headerEl) {
      // Hapus baris badge kartun lama jika ada agar tidak duplikat
      const oldBadgesRow = headerEl.querySelector('.cartoon-badges-row');
      if (oldBadgesRow) oldBadgesRow.remove();

      // HAPUS / HILANGKAN SELURUH BADGE BAWAAN PLATFORM (Verified YouTube, Twitch Badge, dll.)
      const defaultBadgesContainer = headerEl.querySelector('.overlay-badges-container');
      if (defaultBadgesContainer) {
        defaultBadgesContainer.remove();
      }

      // Hapus tag platform icon jika ada di header
      const platformTags = headerEl.querySelectorAll('.overlay-platform-tag');
      platformTags.forEach(tag => tag.remove());

      // Hapus elemen dengan class verified, badge bawaan platform, badge alert, dan tag redeem
      itemEl.querySelectorAll('.platform-official-badge, .twitch-badge, .yt-official-badge, [class*="verified"], .badge-icon, .redeem-tag, .alert-tag, .alert-badge, .overlay-event-badge').forEach(b => {
        try { b.remove(); } catch(e) { b.style.display = 'none'; }
      });

      // Bersihkan jika ada span badge alert kuning (#f59e0b) atau teks ALERT
      itemEl.querySelectorAll('span').forEach(sp => {
        const bg = (sp.style.backgroundColor || sp.style.background || '').toLowerCase();
        const txt = (sp.textContent || '').trim().toUpperCase();
        if (bg.includes('f59e0b') || bg.includes('245, 158, 11') || txt.includes('ALERT') || (sp.classList.contains('badge-icon') && !sp.classList.contains('cartoon-role-badge'))) {
          try { sp.remove(); } catch(e) { sp.style.display = 'none'; }
        }
      });

      const badgesRow = document.createElement('div');
      badgesRow.className = 'cartoon-badges-row';
      let hasBadge = false;

      // Moderator (Twitch & YouTube)
      const isMod = itemEl.dataset.isMod === 'true';
      if (isMod) {
        const b = document.createElement('span');
        b.className = 'cartoon-role-badge role-mod';
        b.innerHTML = '🛡️ Mod';
        badgesRow.appendChild(b);
        hasBadge = true;
      }

      // Subscriber: EKSKLUSIF TWITCH! (YouTube TIDAK PERNAH memunculkan Sub)
      const isSub = isTwitch && itemEl.dataset.isSub === 'true';
      if (isSub) {
        const b = document.createElement('span');
        b.className = 'cartoon-role-badge role-sub';
        b.innerHTML = '★ Sub';
        badgesRow.appendChild(b);
        hasBadge = true;
      }

      // Member: EKSKLUSIF YOUTUBE! (Twitch TIDAK PERNAH memunculkan Member)
      const isMember = isYouTube && itemEl.dataset.isMember === 'true';
      if (isMember) {
        const b = document.createElement('span');
        b.className = 'cartoon-role-badge role-member';
        b.innerHTML = '★ Member';
        badgesRow.appendChild(b);
        hasBadge = true;
      }

      // Broadcaster / Host (Pemilik channel)
      const isHost = itemEl.dataset.isBroadcaster === 'true';
      if (isHost) {
        const b = document.createElement('span');
        b.className = 'cartoon-role-badge role-host';
        b.innerHTML = '👑 Host';
        badgesRow.appendChild(b);
        hasBadge = true;
      }

      // VIP: EKSKLUSIF TWITCH!
      const isVip = isTwitch && itemEl.dataset.isVip === 'true';
      if (isVip) {
        const b = document.createElement('span');
        b.className = 'cartoon-role-badge role-vip';
        b.innerHTML = '💎 VIP';
        badgesRow.appendChild(b);
        hasBadge = true;
      }

      // Pasang HANYA 1 baris badge kartun di bawah nama jika user punya role
      if (hasBadge) {
        headerEl.appendChild(badgesRow);
      }
    }

    // 5. Pastikan teks pesan biasa TIDAK BOLD (500) dan berwarna pekat kontras (#0f172a)
    const textEl = itemEl.querySelector('.overlay-text');
    if (textEl) {
      textEl.style.setProperty('color', '#0f172a', 'important');
      textEl.style.setProperty('font-weight', '500', 'important');
    }
    const msgContentEl = itemEl.querySelector('.overlay-message-content');
    if (msgContentEl) {
      msgContentEl.style.setProperty('color', '#0f172a', 'important');
      msgContentEl.style.setProperty('font-weight', '500', 'important');
    }

    // 6. Alert & Event: Pastikan judul & detail event kontras pekat (#0f172a / #334155) di background putih
    const eventTitleEl = itemEl.querySelector('.overlay-event-title');
    if (eventTitleEl) {
      eventTitleEl.style.setProperty('color', '#0f172a', 'important');
      eventTitleEl.style.setProperty('-webkit-text-fill-color', '#0f172a', 'important');
      eventTitleEl.style.setProperty('font-weight', '700', 'important');
      eventTitleEl.style.setProperty('text-shadow', 'none', 'important');
    }

    if (itemEl.classList.contains('overlay-event') || itemEl.classList.contains('is-redeem') || eventTitleEl) {
      itemEl.style.setProperty('background', 'transparent', 'important');
      itemEl.style.setProperty('color', '#0f172a', 'important');
      if (textEl) {
        textEl.style.setProperty('color', '#334155', 'important');
        textEl.style.setProperty('-webkit-text-fill-color', '#334155', 'important');
        textEl.style.setProperty('font-weight', '500', 'important');
        textEl.style.setProperty('text-shadow', 'none', 'important');
      }
    }
  }

  // Terapkan ke kontainer chat OBS overlay
  const container = document.getElementById('overlayChatContainer');
  if (container) {
    container.querySelectorAll('.overlay-item').forEach(applyThoughtBubble);

    const observer = new MutationObserver((mutations) => {
      mutations.forEach((m) => {
        m.addedNodes.forEach((node) => {
          if (node.nodeType === 1 && node.classList.contains('overlay-item')) {
            applyThoughtBubble(node);
          }
        });
      });
    });
    observer.observe(container, { childList: true });
  }

  console.log('[StreamPulse] Cartoon Thought Bubble initialized: Clean roles, no platform badges!');
})(config, socket);
