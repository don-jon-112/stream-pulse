/* --------------------------------------------------------------------------
   [BAGIAN 2: CUSTOM JS]
   Salin (Copy) kode di bawah ini lalu tempel (Paste) ke kolom "Custom JS"
   di halaman StreamPulse Overlay Studio (http://.../studio).
   -------------------------------------------------------------------------- */

(function initPastelChatBubble(config, socket) {
  // Palet 3 Warna Pastel
  const COLOR_CLASSES = ['bubble-yellow', 'bubble-blue', 'bubble-pink'];
  let colorIndex = 0;

  // Cache user -> warna pastel agar 1 user tetap konsisten warnanya
  const userColorMap = new Map();

  function getPastelColorClass(username) {
    if (!username) {
      const cls = COLOR_CLASSES[colorIndex % COLOR_CLASSES.length];
      colorIndex++;
      return cls;
    }
    const clean = username.toLowerCase();
    if (!userColorMap.has(clean)) {
      const cls = COLOR_CLASSES[colorIndex % COLOR_CLASSES.length];
      colorIndex++;
      userColorMap.set(clean, cls);
    }
    return userColorMap.get(clean);
  }

  // Format pesan Channel Points Redeem
  function formatRedeemText(authorName, rewardTitle, points, messageText) {
    const pts = points ? `${points} Points` : '';
    const rTitle = rewardTitle || 'Channel Points';
    
    if (messageText && messageText.trim().length > 0) {
      // Format dengan teks: "xxx redeemed (redeem event) (xx Points): {text}" dengan style miring/italic
      const ptsPart = pts ? ` (${pts})` : '';
      return `💠 ${authorName} redeemed ${rTitle}${ptsPart}: <i>${messageText.trim()}</i>`;
    } else {
      // Format tanpa teks: "xxx redeemed yyy (xxx Points)"
      const ptsPart = pts ? ` (${pts})` : '';
      return `💠 ${authorName} redeemed ${rTitle}${ptsPart}`;
    }
  }

  // Bersihkan teks redeem Twitch untuk mengekstrak judul & Points
  function parseRedeemDetail(detailText) {
    let reward = 'Channel Points';
    let points = '';
    if (!detailText) return { reward, points };

    const match = detailText.match(/^(.*?)(?:\s*\(([\d.,]+)\s*(?:Points|points|Poin|poin)\))?$/i);
    if (match) {
      if (match[1] && match[1].trim()) reward = match[1].trim();
      if (match[2]) points = match[2].trim();
    } else {
      reward = detailText;
    }
    return { reward, points };
  }

  // Palet warna acak untuk nama YouTube di setiap chat (kontras tinggi & terbaca jelas di atas putih)
  const YT_NAME_COLORS = [
    '#e11d48', // Crimson Red
    '#2563eb', // Royal Blue
    '#059669', // Emerald Green
    '#7c3aed', // Purple / Violet
    '#ea580c', // Warm Orange
    '#db2777', // Magenta Pink
    '#0891b2', // Teal / Cyan Gelap
    '#4f46e5'  // Deep Indigo
  ];

  // Cache untuk mendeteksi & menghapus duplikasi Channel Points Redeem (15 detik)
  const recentRedeems = new Map();

  // Terapkan styling dan manipulasi ke elemen chat item
  function enhanceItem(itemEl) {
    if (!itemEl || itemEl._pastelEnhanced) return;
    itemEl._pastelEnhanced = true;

    const authorEl = itemEl.querySelector('.overlay-author-name');
    let textEl = itemEl.querySelector('.overlay-text');
    const authorName = authorEl ? authorEl.textContent.trim() : '';

    const titleEl = itemEl.querySelector('.overlay-event-title');
    const titleText = titleEl ? titleEl.textContent : '';

    // Deteksi platform secara ketat
    const isTikTok = itemEl.classList.contains('platform-tiktok') || 
                     itemEl.dataset.platform === 'tiktok' ||
                     titleText.includes('TikTok');

    const isYouTube = itemEl.classList.contains('platform-youtube') || 
                      itemEl.dataset.platform === 'youtube' ||
                      titleText.includes('YouTube') ||
                      !!itemEl.querySelector('.yt-official-badge, .yt-member-badge, .yt-mod-badge');

    const isTwitch = !isTikTok && !isYouTube;

    if (authorEl && isYouTube) {
      const randomYtColor = YT_NAME_COLORS[Math.floor(Math.random() * YT_NAME_COLORS.length)];
      authorEl.style.setProperty('color', randomYtColor, 'important');
    }

    // Cek apakah item ini adalah Twitch Channel Points Redeem (HANYA UNTUK TWITCH)
    const isRedeem = isTwitch && (
      itemEl.classList.contains('highlighted') || 
      itemEl.classList.contains('is-redeem') ||
      !!itemEl.querySelector('.redeem-tag')
    );

    // Jika ini adalah event dan overlay belum membuat .overlay-text
    if (!textEl && itemEl.classList.contains('overlay-event')) {
      const rawDetail = titleText;
      const body = itemEl.querySelector('.overlay-body');

      if (isRedeem) {
        // KHUSUS TWITCH REDEEM TANPA TEKS: format "xxx redeemed yyy (xxx Points)"
        let { reward, points } = parseRedeemDetail(rawDetail);
        if (reward.includes('•')) {
          reward = reward.split('•')[1] || reward;
          reward = reward.trim();
        }
        const formatted = formatRedeemText(authorName, reward, points, '');
        if (body) {
          textEl = document.createElement('div');
          textEl.className = 'overlay-text is-redeem bubble-pink';
          textEl.style.fontStyle = 'italic';
          textEl.innerHTML = formatted;
          body.appendChild(textEl);
          itemEl.classList.add('is-redeem');
        }
      } else {
        // UNTUK TIKTOK ATAU YOUTUBE ALERT:
        // SANGAT DILARANG ADA KATA "redeemed"!
        if (body && rawDetail) {
          textEl = document.createElement('div');
          const colorCls = getPastelColorClass(authorName);
          textEl.className = `overlay-text ${colorCls}`;
          textEl.textContent = rawDetail;
          body.appendChild(textEl);
        }
      }
    }

    if (textEl) {
      if (isTikTok) {
        // JAMINAN 100%: TikTok event TIDAK BOLEH mengandung kata "redeemed"
        let currentHtml = textEl.innerHTML || textEl.textContent || '';
        if (currentHtml.includes('redeemed')) {
          currentHtml = currentHtml.replace(/💠\s*/g, '').replace(/\b[\w.-]+\s+redeemed\s+/gi, '').trim();
          textEl.innerHTML = currentHtml;
        }
        textEl.classList.remove('is-redeem');
        itemEl.classList.remove('is-redeem');
        const colorCls = getPastelColorClass(authorName);
        textEl.classList.add(colorCls);
      } else if (isRedeem) {
        itemEl.classList.add('is-redeem');
        textEl.classList.add('bubble-pink');
        textEl.style.fontStyle = 'italic';

        let rawDetail = titleText;
        let cleanText = textEl.textContent.replace(/^"|"$/g, '').trim();

        // Ekstrak pesan bersih untuk mencegah duplikasi (antara event & chat)
        let pureMessage = cleanText;
        if (cleanText.startsWith('💠')) {
          const colonIdx = cleanText.indexOf(': ');
          pureMessage = colonIdx !== -1 ? cleanText.substring(colonIdx + 2).trim() : '';
        }

        // Deduplikasi: Jika redeem untuk author dan pesan ini sudah muncul, hapus bubble duplikat
        const dedupeKey = `${authorName.toLowerCase()}__${pureMessage.toLowerCase()}`;
        if (recentRedeems.has(dedupeKey)) {
          const existingEl = recentRedeems.get(dedupeKey);
          if (existingEl && existingEl.parentNode && existingEl !== itemEl) {
            itemEl.remove();
            return;
          }
        }
        recentRedeems.set(dedupeKey, itemEl);
        setTimeout(() => {
          if (recentRedeems.get(dedupeKey) === itemEl) {
            recentRedeems.delete(dedupeKey);
          }
        }, 15000);

        // Jika teks belum diformat dengan prefix 💠, format sesuai permintaan
        if (!cleanText.startsWith('💠')) {
          let { reward, points } = parseRedeemDetail(rawDetail);
          if (reward.includes('•')) {
            reward = reward.split('•')[1] || reward;
            reward = reward.trim();
          }
          const formatted = formatRedeemText(authorName, reward, points, cleanText);
          textEl.innerHTML = formatted;
        }
      } else {
        // Berikan warna pastel (Yellow, Blue, Pink) untuk chat biasa
        const colorCls = getPastelColorClass(authorName);
        textEl.classList.add(colorCls);

        if (!textEl.querySelector('.overlay-reply-quote')) {
          // Cek format pesan reply (@username ...) jika belum memiliki elemen quote
          const textContent = textEl.textContent.trim();
          const replyMatch = textContent.match(/^@([a-zA-Z0-9_]+)\s*,\s*(.*)$/);
          if (replyMatch) {
            const repliedUser = replyMatch[1];
            const mainMessage = replyMatch[2];
            textEl.innerHTML = `
              <div class="overlay-reply-quote">
                <span class="overlay-reply-icon">↩</span>
                <span class="overlay-reply-author">@${repliedUser}</span>
              </div>
              <span class="overlay-message-content">${mainMessage}</span>
            `;
          }
        }
      }
    }
  }

  // MutationObserver untuk mendeteksi setiap pesan baru yang masuk ke OBS Canvas
  const container = document.getElementById('overlayChatContainer');
  if (container) {
    const observer = new MutationObserver((mutations) => {
      for (const mutation of mutations) {
        for (const node of mutation.addedNodes) {
          if (node.nodeType === 1 && node.classList && node.classList.contains('overlay-item')) {
            enhanceItem(node);
          }
        }
      }
    });

    observer.observe(container, { childList: true });

    // Format item yang mungkin sudah ada saat reload
    Array.from(container.querySelectorAll('.overlay-item')).forEach(enhanceItem);
  }
})(config, socket);
